if ($action === 'generate_all') {
    $period = $_POST['period'] ?? $period;
    try {
        $pdo->beginTransaction();

        // 🔹 1. Get all farmers who have either collections or tea_bag_issues for this month
        $stmt = $pdo->prepare("
            SELECT DISTINCT f.id AS farmer_id, f.area_id
            FROM farmers f
            LEFT JOIN collections c 
                ON c.farmer_id = f.id AND DATE_FORMAT(c.collection_date, '%Y-%m') = :pm
            LEFT JOIN tea_bag_issues t
                ON t.farmer_id = f.id AND DATE_FORMAT(t.issue_date, '%Y-%m') = :pm
            WHERE (c.id IS NOT NULL OR t.id IS NOT NULL)
        ");
        $stmt->execute([':pm' => $period]);
        $rows = $stmt->fetchAll(PDO::FETCH_ASSOC);

        // 🔹 2. Prepare statements
        $checkStmt = $pdo->prepare("SELECT id FROM payments WHERE farmer_id = :f AND period = :p LIMIT 1");
        $insertStmt = $pdo->prepare("
            INSERT INTO payments (farmer_id, period, total_collection, total_payable, saved_amount, paid_amount, created_at, bag_weight)
            VALUES (:f, :p, :tc, :tp, 0, 0, NOW(), 0)
        ");
        $updateStmt = $pdo->prepare("
            UPDATE payments
            SET total_collection = :tc, total_payable = :tp, updated_at = NOW()
            WHERE farmer_id = :f AND period = :p
        ");
        $markCollections = $pdo->prepare("
            UPDATE collections
            SET status = 1
            WHERE farmer_id = :f AND DATE_FORMAT(collection_date, '%Y-%m') = :pm
        ");
        $rateStmt = $pdo->prepare("
            SELECT rate_per_kg 
            FROM area_rates 
            WHERE area_id = :aid AND `year_month` = :pm 
            LIMIT 1
        ");

        // 🔹 3. Process each farmer
        foreach ($rows as $r) {
            $farmerId = (int)$r['farmer_id'];
            $areaId = (int)$r['area_id'];

            // --- Total collection weight ---
            $totStmt = $pdo->prepare("
                SELECT SUM(weight_kg) AS total_wt
                FROM collections
                WHERE farmer_id = :f AND DATE_FORMAT(collection_date, '%Y-%m') = :pm
            ");
            $totStmt->execute([':f'=>$farmerId, ':pm'=>$period]);
            $totRow = $totStmt->fetch(PDO::FETCH_ASSOC);
            $totalCollection = floatval($totRow['total_wt'] ?? 0);

            // --- Total tea bag amount for this month ---
            $teaStmt = $pdo->prepare("
                SELECT COALESCE(SUM(subtotal), 0) AS total_tea_bag
                FROM tea_bag_issues
                WHERE farmer_id = :f AND DATE_FORMAT(issue_date, '%Y-%m') = :pm
            ");
            $teaStmt->execute([':f'=>$farmerId, ':pm'=>$period]);
            $tea_bag_total = floatval($teaStmt->fetchColumn() ?? 0);

            // --- Area rate ---
            $rateStmt->execute([':aid'=>$areaId, ':pm'=>$period]);
            $rrow = $rateStmt->fetch(PDO::FETCH_ASSOC);
            $rate = $rrow ? floatval($rrow['rate_per_kg']) : 0.0;

            $bag_weight = 0.0;
            $balance_weight = max(0, $totalCollection - $bag_weight);
            $total_pay = round($balance_weight * $rate, 2);

            // --- Optionally deduct tea bag cost ---
            $total_payable = max(0, $total_pay - $tea_bag_total);

            // --- Insert or update payment record ---
            $checkStmt->execute([':f'=>$farmerId, ':p'=>$period]);
            if ($checkStmt->rowCount() > 0) {
                $updateStmt->execute([':tc'=>$totalCollection, ':tp'=>$total_payable, ':f'=>$farmerId, ':p'=>$period]);
            } else {
                $insertStmt->execute([':f'=>$farmerId, ':p'=>$period, ':tc'=>$totalCollection, ':tp'=>$total_payable]);
            }

            // --- Mark collections as processed ---
            $markCollections->execute([':f'=>$farmerId, ':pm'=>$period]);
        }

        $pdo->commit();
        json_exit(['status'=>true, 'message'=>'Payments generated/updated successfully (including tea bag farmers).']);
    } catch (Exception $e) {
        $pdo->rollBack();
        json_exit(['status'=>false, 'message'=>'Error: '.$e->getMessage()]);
    }

            ]);

<script>
$(document).on('click', '.pay-link', function(e) {
  e.preventDefault(); // stop normal link behavior

  const link = $(this);
  const id = link.data('id');
  const farmerId = link.data('farmer-id');
  const period = link.data('period');
  const full = link.data('full');
  const full1 = link.data('full1');
  const lmbalance_total = link.data('lmbalance_total');
  const href = link.attr('href');

  if (!confirm('Are you sure you want to mark this as paid and print the slip?')) return;

  // open slip in new tab for printing
  window.open(href, '_blank');

  // update payment status in backend
  $.ajax({
    url: 'update_payment_status.php',
    type: 'POST',
    data: { id: id, farmer_id: farmerId, period: period, full: full, full1: full1, lmbalance_total: lmbalance_total },
    
	success: function(res) {
  try {
    const json = (typeof res === 'string') ? JSON.parse(res) : res;
    if (json.success) {
      // queue reload before showing alert
      setTimeout(() => location.reload(), 500);
      alert('Payment updated successfully!');
    } else {
      alert('Error: ' + json.message);
    }
  } catch (e) {
    console.error('Parse error:', e, res);
    alert('Unexpected response from server.');
  }
},
	
    error: function() {
      alert('Server error occurred while updating.');
    }
  });
});
</script>

$(document).on('click', '.lmbalance-btn', function() {
  const btn = $(this);
  const paymentId = btn.data('payment');
  const amount = parseFloat(btn.data('amount')) || 0;

      alert('LMBalance saved successfully!');

      });

    }
  }, 'json').fail(() => {
    alert('Server error.');
    btn.prop('disabled', false).text('LMBalance');
  });
});
